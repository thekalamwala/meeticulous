# CRM Lists (cold contact pool) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a "Lists" cold-contact pool to the admin Sales CRM — import bulk CSV contacts that live as `status='list'` leads (outside the pipeline), cold-email them, and manually promote (`list→new`) or demote (`→list`) between List and Pipeline.

**Architecture:** Cold contacts ARE `leads`, marked by a new lifecycle status `list` + a `list_id` grouping column. `lifecycleRules.js` centralizes the state so the engine skips it (NO_AUTO), desk/reports/follow-ups exclude it (CLOSED), and no kanban board renders it. Promote/demote are ordinary `changeStatus` calls. Cold email reuses the existing gated campaign/bulk send path. No parallel store, no new send stack.

**Tech Stack:** Node/Express + Supabase (server), React 18 + Vite + react-router-dom v6 (client), Jest (server tests), Vitest + @testing-library/react (client tests).

**Spec:** [docs/superpowers/specs/2026-09-11-crm-lists-cold-pool-design.md](../specs/2026-09-11-crm-lists-cold-pool-design.md)

## Global Constraints

- **Two databases:** every migration runs on **staging (Tokyo `STAGING_PROJECT_REF`) → prod (Mumbai `PROD_PROJECT_REF`)**. Keep code-side schema fallback (destructure/strip on missing-column) until confirmed applied on both.
- **Never regress an existing lead:** `upsertLead` merge already skips `status`/`source`/`owner_admin_id` (leadService.js:72). Cold status is set only on CREATE.
- **Cold prospects must not fire Meta CAPI Lead events** — pass `skipMetaLead: true` on every list import.
- **Sends stay gated:** cold email goes through the existing `crmBulk` → `messageService.enqueue` (kill-switch `crm_admin_sends`, DNC, opt-out, frequency cap). Build no new send path.
- **UI rules:** reuse existing `components/admin/ui` (`Table`, `Card`, `Btn`, `Badge`), `TagChip`, tokens; UI term is "List"/"Lists"; no new colors. Lists is a **flat table, never a kanban board**.
- **Deploy gate:** `cd client && npx vitest run` must pass before any client PR; `cd server && npm test` before server PR.
- **DB status keys are never renamed** — `list` is additive to the existing CHECK.

```js
// server/services/crm/listService.js
'use strict';
/**
 * listService.js — CRM Lists (cold contact pools). A list groups cold leads
 * (status='list', list_id=<this>). Promotion/sending reuse the lead pipeline.
 */
const { supabaseAdmin } = require('../../config/supabase');

async function createList({ name, source = null, consentBasis = null, description = null, createdBy = '' }) {
  const clean = String(name || '').trim();
  if (!clean) throw new Error('list name required');
  const { data, error } = await supabaseAdmin.from('lead_lists')
    .insert({ name: clean, source, consent_basis: consentBasis, description, created_by: createdBy })
    .select('*').single();
  if (error) throw error;
  return data;
}

async function getList(id) {
  const { data } = await supabaseAdmin.from('lead_lists').select('*').eq('id', id).maybeSingle();
  return data || null;
}

// Lists newest-first, each with its live cold-lead count. Counts are derived
// (one head-count query per list) — no stored counter to drift. Fine at this scale.
async function listLists() {
  const { data, error } = await supabaseAdmin.from('lead_lists').select('*').order('created_at', { ascending: false });
  if (error) throw error;
  const lists = data || [];
  await Promise.all(lists.map(async (l) => {
    const { count } = await supabaseAdmin.from('leads')
      .select('id', { count: 'exact', head: true }).eq('list_id', l.id).eq('status', 'list');
    l.cold_count = count || 0;
  }));
  return lists;
}

module.exports = { createList, getList, listLists };
```

```js
const KNOWN_SOCIALS = ['twitter', 'x', 'youtube', 'facebook'];

const importCommit = async (req, res) => {
  try {
    const rows = Array.isArray(req.body?.rows) ? req.body.rows : [];
    // Resolve the target list: an existing id, or create one from listName.
    let listId = req.body?.listId || null;
    if (!listId && req.body?.listName) {
      const created = await listService.createList({ name: req.body.listName, source: req.body.source || 'import', createdBy: actor(req) });
      listId = created.id;
    }
    let created = 0, merged = 0, skipped = 0;
    for (const r of rows) {
      if (!r.phone && !r.email) { skipped++; continue; }
      // Stash unknown/extra socials so nothing is lost.
      const attributes = {};
      for (const k of KNOWN_SOCIALS) if (r[k]) attributes[k === 'x' ? 'twitter' : k] = r[k];
      const patch = {
        name: r.name || '', phone: r.phone || '', whatsapp_phone: r.whatsapp || r.phone || '',
        email: r.email || '', business_name: r.business || '', profession: r.profession || '',
        website: r.website || '', linkedin: r.linkedin || '', instagram: r.instagram || '',
        city: r.city || '', tags: Array.isArray(r.tags) ? r.tags : (r.tags ? String(r.tags).split(/[,|]/).map((t) => t.trim()).filter(Boolean) : []),
      };
      if (Object.keys(attributes).length) patch.attributes = attributes;
      if (listId) { patch.status = 'list'; patch.list_id = listId; }   // cold on create; merge won't regress an existing lead

        if (isNew) created++; else merged++;
      } catch (_) { skipped++; }
    }
    res.json({ created, merged, skipped, listId });

};
```

```jsx
function parseCsv(text) {
  const rows = []; let row = [], field = '', q = false;
  for (let i = 0; i < text.length; i++) {
    const c = text[i];
    if (q) { if (c === '"' && text[i + 1] === '"') { field += '"'; i++; } else if (c === '"') q = false; else field += c; }
    else if (c === '"') q = true;
    else if (c === ',') { row.push(field); field = ''; }
    else if (c === '\n' || c === '\r') { if (field !== '' || row.length) { row.push(field); rows.push(row); row = []; field = ''; } if (c === '\r' && text[i + 1] === '\n') i++; }
    else field += c;
  }
  if (field !== '' || row.length) { row.push(field); rows.push(row); }
  return rows;
}

          <Btn size="sm" onClick={commit} disabled={busy || !count}>{busy ? 'Importing…' : `Import ${count}`}</Btn>
        </div>
      </Card>
    </div>
  );
}
```
